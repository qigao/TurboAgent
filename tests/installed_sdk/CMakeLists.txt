cmake_minimum_required(VERSION 3.25)
project(TurboAgentInstalledConsumers LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_SKIP_BUILD_RPATH TRUE)
if(MSVC)
  add_compile_options(/utf-8 /Zc:preprocessor /Zc:__cplusplus)
  add_compile_options($<$<COMPILE_LANGUAGE:C>:/experimental:c11atomics>)
endif()

# Resolve installed SDKs and the producer's installed ports only. Never add
# the TurboAgent source/build directories or consult a user package registry.
set(CMAKE_FIND_USE_PACKAGE_REGISTRY FALSE)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY FALSE)
set(CMAKE_FIND_USE_CMAKE_ENVIRONMENT_PATH FALSE)
set(CMAKE_FIND_USE_PACKAGE_ROOT_PATH FALSE)
set(CMAKE_FIND_USE_CMAKE_SYSTEM_PATH FALSE)
set(CMAKE_FIND_USE_SYSTEM_ENVIRONMENT_PATH FALSE)
set(CMAKE_PREFIX_PATH "$ENV{TURBOAGENT_CONSUMER_PORTS_ROOT}")
foreach(root SALTS_ROOT SALTS_UTILS_ROOT CHTTP_ROOT TURBOWASM_ROOT
             TURBOSCRIPT_ROOT PRAKTOR_ROOT)
  if(NOT "$ENV{${root}}" STREQUAL "")
    list(APPEND CMAKE_PREFIX_PATH "$ENV{${root}}")
  endif()
endforeach()
find_package(TurboAgent CONFIG REQUIRED
  PATHS "$ENV{TURBOAGENT_CONSUMER_SDK_ROOT}" NO_DEFAULT_PATH)
message(STATUS "Installed consumer TurboAgent_DIR=${TurboAgent_DIR}")
set(sdk_root "$ENV{TURBOAGENT_CONSUMER_SDK_ROOT}")
if(TARGET TurboAgent::LangChain OR TARGET LangChain::LangChain)
  message(FATAL_ERROR "Retired LangChain target escaped into the installed package")
endif()
foreach(module RuntimeCore RuntimeData Runtime RuntimeTools Provider AgentCompiler Agent Harness)
  get_target_property(imported TurboAgent::${module} IMPORTED)
  if(NOT imported)
    message(FATAL_ERROR "TurboAgent::${module} must be an installed imported target")
  endif()
  get_target_property(include_dirs TurboAgent::${module} INTERFACE_INCLUDE_DIRECTORIES)
  foreach(include_dir IN LISTS include_dirs)
    cmake_path(IS_PREFIX sdk_root "${include_dir}" NORMALIZE in_sdk)
    if(NOT in_sdk)
      message(FATAL_ERROR "TurboAgent::${module} exposes headers outside the relocated SDK: ${include_dir}")
    endif()
  endforeach()
endforeach()

enable_testing()
add_executable(installed_public_c11 ../../harness/tests/test_turbo_agent_public_api.c)
add_executable(installed_harness_c11 ../../harness/tests/test_turbo_agent_harness.c)
add_executable(installed_remote_c11 ../../agent/tests/test_turbo_agent_runtime_remote_api.c)
add_executable(installed_cpp17 consumer.cpp)
foreach(consumer installed_public_c11 installed_harness_c11 installed_remote_c11 installed_cpp17)
  target_link_libraries(${consumer} PRIVATE
    TurboAgent::Harness TurboAgent::Agent TurboAgent::AgentCompiler
    TurboAgent::Runtime TurboAgent::Provider Salts::TinyTest)
  add_test(NAME ${consumer} COMMAND ${consumer})
  set_tests_properties(${consumer} PROPERTIES TIMEOUT 60)
endforeach()

# A Provider-only consumer must not inherit graph/storage Runtime dependencies.
get_target_property(provider_links TurboAgent::Provider INTERFACE_LINK_LIBRARIES)
if("TurboAgent::Runtime" IN_LIST provider_links)
  message(FATAL_ERROR "Provider must depend on event/data contracts, not durable Runtime")
endif()
add_executable(installed_provider_c11 provider.c)
target_link_libraries(installed_provider_c11 PRIVATE TurboAgent::Provider)
add_test(NAME installed_provider_c11 COMMAND installed_provider_c11)

add_executable(installed_job_c11 ../../harness/tests/test_turbo_agent_job.c)
target_link_libraries(installed_job_c11 PRIVATE TurboAgent::Harness Salts::TinyTest)
add_test(NAME installed_job_c11 COMMAND installed_job_c11)
set_tests_properties(installed_job_c11 PROPERTIES TIMEOUT 60)

add_executable(installed_job_harness_c11 ../../harness/tests/test_turbo_agent_harness_compiled.c)
target_link_libraries(installed_job_harness_c11 PRIVATE TurboAgent::Harness Salts::TinyTest)
add_test(NAME installed_job_harness_c11 COMMAND installed_job_harness_c11)
set_tests_properties(installed_job_harness_c11 PROPERTIES TIMEOUT 60)

add_executable(installed_job_files_c11 ../../harness/tests/test_turbo_agent_job_file_verifier.c)
target_link_libraries(installed_job_files_c11 PRIVATE TurboAgent::Harness Salts::TinyTest)
add_test(NAME installed_job_files_c11 COMMAND installed_job_files_c11)
set_tests_properties(installed_job_files_c11 PROPERTIES TIMEOUT 60)
